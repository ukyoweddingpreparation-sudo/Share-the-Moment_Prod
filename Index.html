    async function startUpload() {

      const tableNo = tableNoInput.value; 

      const userName = userNameInput.value.trim();


      if (selectedFiles.length === 0 || !tableNo) return;


      uploadBtn.style.display = 'none';

      addBtn.style.display = 'none';

      resetBtn.style.display = 'none';

      progressContainer.style.display = 'block';


      const totalPhotos = selectedFiles.length;

      let compressedCount = 0;

      let uploadedPhotoCount = 0;

      let targetFolderId = null;


      try {

        updateProgress(0, `最適化中 (0/${totalPhotos}枚)...`);


        const fileDataArray = await Promise.all(

          selectedFiles.map(file => 

            resizeAndCompressImage(file, 2000, 0.85, () => {

              compressedCount++;

              const currentPercent = (compressedCount / totalPhotos) * 30;

              updateProgress(currentPercent, `最適化中 (${compressedCount}/${totalPhotos}枚)...`);

            })

          )

        );


        const CHUNK_SIZE = 3;

        const totalChunks = Math.ceil(fileDataArray.length / CHUNK_SIZE);


        for (let i = 0; i < fileDataArray.length; i += CHUNK_SIZE) {

          const chunk = fileDataArray.slice(i, i + CHUNK_SIZE);

          const currentChunkIndex = Math.floor(i / CHUNK_SIZE) + 1;

          const currentBatchSize = chunk.length; 


          if (currentChunkIndex === totalChunks) {

            updateProgress(100, `送信完了 (${totalPhotos}/${totalPhotos}枚)`);

            setTimeout(() => {

              progressContainer.style.display = 'none';

              confirmScreen.style.display = 'none';

              thankyouScreen.style.display = 'block';

              window.scrollTo(0, 0);

            }, 300);


            sendBatchToGas(chunk, tableNo, userName, targetFolderId).catch(err => {

              console.error('Background upload error:', err);

            });

          } else {

            const result = await sendBatchToGas(chunk, tableNo, userName, targetFolderId);

            if (!result.success) {

              throw new Error(result.error);

            }

            

            targetFolderId = result.folderId;

            uploadedPhotoCount += currentBatchSize;


            const sendPercent = 30 + (uploadedPhotoCount / totalPhotos) * 65;

            updateProgress(sendPercent, `送信中 (${uploadedPhotoCount}/${totalPhotos}枚)...`);

          }

        }


      } catch (err) {

        progressContainer.style.display = 'none';

        showError('送信に失敗しました: ' + (err.message || err));

      }

    }


    function resizeAndCompressImage(file, maxSideLength, quality, onProgress) {

      return new Promise((resolve, reject) => {

        const reader = new FileReader();

        reader.onload = (e) => {

          const img = new Image();

          img.onload = () => {

            let width = img.width;

            let height = img.height;


            if (width > height) {

              if (width > maxSideLength) {

                height = Math.round((height * maxSideLength) / width);

                width = maxSideLength;

              }

            } else {

              if (height > maxSideLength) {

                width = Math.round((width * maxSideLength) / height);

                height = maxSideLength;

              }

            }


            const canvas = document.createElement('canvas');

            canvas.width = width;

            canvas.height = height;

            const ctx = canvas.getContext('2d');

            ctx.drawImage(img, 0, 0, width, height);


            const dataUrl = canvas.toDataURL('image/jpeg', quality);

            

            if (onProgress) onProgress();


            resolve({

              base64Data: dataUrl,

              fileName: file.name

            });

          };

          img.onerror = (err) => reject(err);

          img.src = e.target.result;

        };

        reader.onerror = (err) => reject(err);

        reader.readAsDataURL(file);

      });

    }
